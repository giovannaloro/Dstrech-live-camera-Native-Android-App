cmake_minimum_required(VERSION 3.22.1)
project(dstretch_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../..")
add_library(dstretch SHARED bridge.cpp "${CORE_ROOT}/native/rgba_adapter.cpp" "${CORE_ROOT}/dstrech.cpp")
target_include_directories(dstretch PRIVATE "${CORE_ROOT}" "${CORE_ROOT}/native")
target_compile_options(dstretch PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
target_link_options(dstretch PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(dstretch PRIVATE jnigraphics)
