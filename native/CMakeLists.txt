cmake_minimum_required(VERSION 3.16)
project(dstretch_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(SANITIZE "Enable address/undefined sanitizers" OFF)
if(SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(core STATIC ../dstrech.cpp rgba_adapter.cpp)
target_include_directories(core PUBLIC .. .)
target_compile_options(core PRIVATE -Wall -Wextra -ffp-contract=off)
add_executable(adapter_test tests/adapter_test.cpp ../helpers.cpp)
target_link_libraries(adapter_test PRIVATE core)
add_executable(desktop ../main.cpp ../helpers.cpp)
target_link_libraries(desktop PRIVATE core)
enable_testing()
add_test(NAME strides_rotation_errors COMMAND adapter_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME desktop_adapter_parity COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/compare.py $<TARGET_FILE:desktop> $<TARGET_FILE:adapter_test>)
